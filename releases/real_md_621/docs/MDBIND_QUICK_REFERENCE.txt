import torch
data = torch.load('data/mdbind_processed/A004B.pt', map_location='cpu', weights_only=False)

# Main objects
graph = data['graph']               # torch_geometric.data.Data
metadata = data['metadata']         # Dict with processing info
temporal_slices = data['temporal_slices']  # Per-replicate snapshots

================================================================================
GRAPH STRUCTURE
================================================================================

graph.x          → [num_residues, 5]
  [0] RMSF (flexibility, Ångströms)
  [1] SASA mean (surface area)
  [2] SASA std
  [3] pocket_flag (1=contacts ligand)
  [4] ligand_contact_freq (0-1)

graph.edge_index → [2, num_edges]
  Sparse residue-residue contact edges

graph.edge_attr  → [num_edges, 5]
  [0] static_contact (binary)
  [1] dynamic_contact_occupancy (0-1)
  [2] dccm (cross-correlation, -1 to 1)
  [3] ca_dist_mean (nanometers)
  [4] ca_dist_std (nanometers)

graph.num_nodes  → int (number of residues)

================================================================================
PROTEIN METADATA
================================================================================

data['residue_names']         → List[str] (e.g., ['ALA', 'GLY', 'VAL', ...])
data['residue_numbers']       → List[int] (PDB resSeq)
data['chain_ids']             → List[str] (e.g., ['A', 'A', 'A', 'B', 'B', ...])

data['node_feature_names']    → ['rmsf', 'sasa_mean', 'sasa_std', 'pocket_flag', 'ligand_contact_freq']
data['edge_feature_names']    → ['static_contact', 'dynamic_contact_occupancy', 'dccm', 'ca_dist_mean', 'ca_dist_std']

================================================================================
LIGAND INFORMATION
================================================================================

data['ligand_residue_names']  → List[str] (e.g., ['GDP'] or ['CMG', 'TPO'])
data['ion_residue_names']     → List[str] (metal ions present)

metadata['num_ligand_residues']  → int (count of ligand residues)
metadata['ligand_contact_cutoff_nm']  → 0.45 (contact threshold)

IMPORTANT: Only 3-letter PDB codes stored, NO SMILES/structures

================================================================================
TEMPORAL INFORMATION
================================================================================

# 10 replicates per protein
for slice_dict in data['temporal_slices']:
    rep = slice_dict['replicate']           # 1-10
    frames = slice_dict['num_frames']       # frames in this replicate
    node_feats = slice_dict['node_features']    # [num_residues, 5]

    # Dense edge features per replicate
    edge_info = slice_dict['edge_features_dense']
    edge_info['pair_indices']               # all pairs
    edge_info['static_contact']             # binary
    edge_info['dynamic_contact_occupancy']  # 0-1
    edge_info['ca_dist_mean']               # mean distance
    edge_info['ca_dist_std']                # std distance
    edge_info['dccm']                       # correlation matrix

================================================================================
FINDING BINDING POCKET
================================================================================

# Method 1: Use pocket_flag (binary)
pocket_mask = data['graph'].x[:, 3] > 0
pocket_residues = [
    f"{data['residue_names'][i]}{data['residue_numbers'][i]}"
    for i in range(len(data['residue_names'])) if pocket_mask[i]
]

# Method 2: Use contact frequency (continuous)
contact_freq = data['graph'].x[:, 4]
contact_threshold = 0.1  # 10% of frames
pocket_residues_cont = [
    f"{data['residue_names'][i]}{data['residue_numbers'][i]}: {contact_freq[i]:.2%}"
    for i in range(len(data['residue_names'])) if contact_freq[i] > contact_threshold
]
